Object Student trong JavaScript

Định nghĩa cấu trúc đối tượng Sinh viên gồm các thuộc tính: Name, NumberID, Gender.

// Định nghĩa Object Student:
function Student(name, numberID, gender) {
  this.Name = name;
  this.NumberID = numberID;
  this.Gender = gender;
}

// Khởi tạo:
const student1 = new Student("Nguyễn Nhật Huy", "6651071028", "Nam");
